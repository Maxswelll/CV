សូមស្វាគមន៍មកកាន់ ប្រាសាទអង្គរវត្ត

មរតកវប្បធម៌ដ៏អស្ចារ្យនៃប្រជាជាតិខ្មែរ ដែលបញ្ចេញសិល្បៈ និងប្រវត្តិសាស្ត្រដ៏សម្បើម។

អំពីប្រាសាទអង្គរវត្ត

ប្រាសាទអង្គរវត្ត គឺជាប្រាសាទសាសនាផ្កាយនៃអាណាចក្រខ្មែរ ដែលស្ថិតនៅក្នុងខេត្តសៀមរាប ប្រទេសកម្ពុជា។ វាជាប្រាសាទដែលធំជាងគេបំផុតនៅលើពិភពលោក ហើយត្រូវបានចាត់ទុកថា ជាស្នាដៃស្ថាបត្យកម្មដ៏អស្ចារ្យមួយនៃសតវត្សទី ១២។

ប្រាសាទនេះត្រូវបានសង់ឡើងដោយព្រះបាទសូរ្យវរ្ម័នទី២ ដើម្បីឧទ្ទិសដល់ព្រះវិស្ណុ។ វាបង្ហាញពីការរួមបញ្ចូលគ្នារវាងសាសនាហិណ្ឌូ និងព្រះពុទ្ធសាសនា ក្នុងវប្បធម៌ខ្មែរ។

ប្រវត្តិនៃប្រាសាទអង្គរវត្ត

ប្រាសាទអង្គរវត្ត ត្រូវបានសង់ឡើងប្រហែលឆ្នាំ ១១១៣ ដល់ ១១៥០ ម.ស. ដោយព្រះបាទសូរ្យវរ្ម័នទី២។ វាជារូបសញ្ញានៃអំណាច និងសេចក្ដីជឿរបស់អាណាចក្រខ្មែរ។

បន្ទាប់ពីសម័យអង្គរធ្លាក់ចុះ ប្រាសាទនេះត្រូវបានបោះបង់ចោលសម្រាប់រយៈពេលជាច្រើនសតវត្សរ៍ ទាល់តែអ្នកស្រាវជ្រាវបារាំងឃើញឡើងវិញក្នុងសតវត្សទី ១៩។

ស្ថាបត្យកម្មនៃប្រាសាទអង្គរវត្ត

ស្ថាបត្យកម្មនៃប្រាសាទអង្គរវត្តមានលក្ខណៈពិសេស និងមានការរៀបចំយ៉ាងម៉ត់ចត់។ វាមានប្រាសាទចម្បង៥ ដែលតំណាងឲ្យភ្នំមេរី ដែលជាទីកន្លែងស្នាក់នៅរបស់ទេវតា។

បន្ទប់ចម្លាក់ និងសិលាចារឹកក្នុងប្រាសាទបង្ហាញពីរឿងរ៉ាវក្នុងរឿងរាមាយណា និងមហាបារៈតា។ ការចម្លាក់លើជញ្ជាំងគឺមានភាពលម្អិត និងជាប្រភេទសិល្បៈដ៏អស្ចារ្យ។

ការវិភាគ និងមតិយោបល់ពីភ្ញៀវទេសចរ

ភ្ញៀវទេសចរពីទូទាំងពិភពលោកបានអះអាងថា ប្រាសាទអង្គរវត្តជាទីកន្លែងមួយដែលមិនអាចខកខានបាន។ ពួកគេគោរពសរសើរចំពោះសិល្បៈដ៏ស្រស់ស្អាត និងសន្តិភាពនៅទីនោះ។

ពេលព្រះអាទិត្យរះ ឬលិច ខាងក្រោយប្រាសាទ គឺជាភាពអស្ចារ្យដូចជាសុបិន្ត។ ភ្ញៀវជាច្រើនថា វាជាបទពិសោធន៍មួយដែលធ្វើឲ្យចិត្តពេញដោយការភ្ញាក់ផ្អើល និងកិត្តិយសចំពោះប្រវត្តិសាស្ត្រខ្មែរ។

អង្គរវត្តមិនត្រឹមតែជាប្រវត្តិសាស្ត្រទេ ប៉ុន្តែជាស្នាដៃសិល្បៈដែលបង្ហាញពីភាពខ្ពង់ខ្ពស់នៃវប្បធម៌ខ្មែរ។

រូបភាពអង្គរវត្ត

ទីតាំងនៃប្រាសាទអង្គរវត្ត
